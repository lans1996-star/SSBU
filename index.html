        function generateGameplan(player, opponent) {

            let plan = `<div style="margin-bottom:0.75rem;"><strong style="color:var(--text-title);">${player.name}:</strong> `;

            let pArch = player.arch.toLowerCase(); let oArch = opponent.arch.toLowerCase();


            if (pArch.includes("zoner") || pArch.includes("trapper")) {

                plan += `Juega al desgaste. Mantén a ${opponent.name} a raya y dicta el ritmo. Usa tu ${player.oos.split('.')[0]} OoS si se acerca demasiado. `;

            } else if (pArch.includes("rushdown") || pArch.includes("combo")) {

                if (oArch.includes("zoner") || oArch.includes("trapper")) plan += `Paciencia. Camina y usa parries para romper su muro defensivo. Cero saltos predecibles. `;

                else plan += `Presión constante. Capitaliza sus errores y llévalo rápido al ledge. `;

            } else if (pArch.includes("swordfighter")) {

                plan += `Usa la punta de tu espada (espaciado perfecto) para crear un muro impenetrable y evitar castigos OoS de ${opponent.name}. `;

            } else if (pArch.includes("heavyweight")) {

            } else {

                plan += `Adapta tu juego a los hábitos de ${opponent.name}. Castiga sus opciones repetitivas. `;

            }

            plan += `</div>`;

            return plan;

        }

            const vText = document.getElementById("verdictText");

            let bWins = pB.best.some(m => pA.name.includes(m) || pA.arch.includes(m)) || pB.worst.some(m => pB.name.includes(m) || pB.arch.includes(m));

                if(tiers[pA.tier] > tiers[pB.tier] + 2) { aWins = true; bWins = false; }

                else if(tiers[pB.tier] > tiers[pA.tier] + 2) { bWins = true; aWins = false; }

            }

            } else {

                vBox.style.borderColor = "var(--even)"; vHeader.style.color = "var(--even)";

                vHeader.textContent = `⚖️ Matchup Igualado (Even)`;

                verdictHtml = `<strong>Razón Técnica:</strong> Parejo. Depende totalmente de quién gestione mejor el neutral y la presión OoS.`;

            }

        function renderGrid() {

            const query = document.getElementById("searchInput").value.toLowerCase();

            const sortMode = document.getElementById("sortSelect").value;

            const grid = document.getElementById("rosterGrid");

            grid.innerHTML = "";

            

            let filtered = [...db];

            

            // Filtrado

            if(activeMainFilter) {

                filtered = filtered.filter(f => f.name === activeMainFilter);

            } else if (query) {

                filtered = filtered.filter(f => f.name.toLowerCase().includes(query) || f.arch.toLowerCase().includes(query));

            }


            // Ordenamiento

            if (sortMode === 'tier') {

                filtered.sort((a, b) => {

                    let wA = tierWeights[a.tier] || 99;

                    let wB = tierWeights[b.tier] || 99;

                    if(wA !== wB) return wA - wB;

                    return a.name.localeCompare(b.name);

                });

            } else {

                filtered.sort((a, b) => a.name.localeCompare(b.name));

            }

            

            if(filtered.length === 0) {

                grid.innerHTML = `<div style="grid-column: 1/-1; text-align: center; padding: 2rem; color: var(--text-body);">No se encontraron luchadores.</div>`;

                return;

            }


            filtered.forEach(f => {

                const card = document.createElement("div");

                const isMain = myMains.includes(f.name);

                card.className = `ssbu-card ${isMain ? 'is-main' : ''}`;

                card.onclick = () => openModal(f);

                

                const baseTier = f.tier.charAt(0);

                const tColor = tierColors[baseTier] || "var(--text-body)";

                

                card.innerHTML = `

                    <div class="ssbu-card-star">⭐</div>

                    <div class="ssbu-avatar" style="background: linear-gradient(135deg, ${tColor}, var(--bg-panel)); border: 1px solid ${tColor};">${getAvatarInitials(f.name)}</div>

                    <div class="ssbu-card-info">

                        <div class="ssbu-card-name">${f.name}</div>

                        <div class="ssbu-card-arch" style="color:${tColor};">Tier ${f.tier}</div>

                    </div>

            });

        }
